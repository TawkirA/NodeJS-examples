5. Which approach is better and why?
    const jamesBond = {
        firstName: "Daniel",
        lastName: "Craig",
        getFullName: function () {
           return `${this.firstName} ${this.lastName}`.trim();
        }
   };
   jamesBond.getFullName();

	OR
  
   class Person {
       constructor(firstName, lastName) {
           this.firstName = firstName;
           this.lastName = lastName;
       }
    }
    Person.prototype.getFullName = function () {
        return `${this.firstName} ${this.lastName}`.trim();
    };
    const jamesBond = new Person("Daniel", "Craig");
    jamesBond.getFullName();

Ans - Second approach is better since in the first approach, a closure is maintained for each copy of the object containing getFullName method. While in the second approach, the method is registered in the prototype rather than in every object. Thus, it is a more memory efficient approach.

7. write a function in JavaScript to merge two objects without overwriting existing properties?

Ans - const mergeObjects = (obj1, obj2) => ({ ...obj1, ...obj2 });
	This function utilizes the spread (...) operator within an object literal to merge the properties of obj1 and obj2. The order is important; properties of obj2 will overwrite properties of obj1 with the same name. This one-liner is effective for creating a new object that contains the combined properties of both objects without modifying the original objects.

10. Difference between object.create and object creation with the new operator?

Ans - 1. Objects created using object literal i.e. with object.create
	 Since these are singletons, a change to an object persists throughout the script. A change in one instance will affect all the instances of the object.

	 var student = {
      		name: "Ravi"
   	}
	var newStudent = student;
   		document.write("student name = " + student.name); // Ravi
   		document.write("new student name = " + newStudent.name); // Ravi
	newStudent.name = "kumar";
   		document.write("student name = " + student.name); // Kumar
   		document.write("new student name = " + newStudent.name); //Kumar

	2. New operator
	Object-defined with a function constructor lets you have multiple instances of that object. This means a change made to one instance, will not affect other instances.
	
	var stu = function() {
      		this.name= "Ravi"
	}
	var newStudent = new stu();
	var newStudent2 = new stu();
   	document.write("student name = " + newStudent.name); // Ravi
   	document.write("new student name = " + newStudent2.name); // Ravi
	newStudent2.name = "kumar";
   	document.write("student name = " + student.name); // Ravi
   	document.write("new student name = " + newStudent.name); //Kumar
	This is due to the important difference that the 2nd method actually runs constructor code, whereas Object.create will not execute the constructor code.
	
	function Dog(){
    		this.eat = 'meat';
	};

	Dog.prototype.child = 'Pups';
	var maddie = new Dog();
	var buddy = Object.create(Dog.prototype);

	//Using Object.create()

    let slicedArr = [];

    for(let i=0; i<arr.length; i++) {

    let max = arr[0];

// Q. Can you write a concise function in JavaScript to implement a simple debounce function that delays the execution of a given function until after a specified time interval has passed without additional calls?
// Sol.
function debounce(func, delay) {
    let debounceTimeout;
    return (...args) => {
        clearTimeout(debounceTimeout);
        debounceTimeout = setTimeout(() => func(...args), delay)
    }
}

const delayedLog = debounce((text) => console.log(text), 1000);
delayedLog('Hello');
delayedLog('World');

// Q. Can you write a throttle function in JavaScript to implement a simple throttle function that limits the execution of a given function to once every specified time interval?
// Sol.
function throttle(func, delay) {
    let throttled = false;
    return (...args) => {
        if (!throttled) {
            func(...args);
            throttled = true;
            setTimeout(() => throttled = false, delay)
        }
    }
}

const throttledLog = throttle((text) => console.log(text), 1000);
throttledLog('India');
throttledLog('USA');
